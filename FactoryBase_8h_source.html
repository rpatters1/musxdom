138public:
139 static EnumClass xmlToEnum(const FromClass& value)
140 {
141 auto it = XmlEnumMapping<EnumClass>::mapping.find(value);
142 if (it != XmlEnumMapping<EnumClass>::mapping.end()) {
143 return it->second;
144 }
145 if constexpr (!IgnoreUnknown) {
146 std::string msg = [value]() {
147 if constexpr (std::is_arithmetic_v<FromClass>) {
148 return "Invalid enum value from xml: " + std::to_string(value);

183template <typename T>
184struct FieldPopulator : public FactoryBase
185{
186 static void populateField(const std::shared_ptr<T>& instance, const XmlElementPtr& fieldElement)
187 {
188 auto it = elementXref().find(fieldElement->getTagName());
189 if (it != elementXref().end()) {
190 std::get<1>(*it)(fieldElement, instance);
191 } else {
192 const bool requireFields = [instance]() {
193 if constexpr (std::is_base_of_v<EnigmaBase, T>) {
194 return instance->requireAllFields();
195 } else {
196 return true;
197 }
198 }();
199 if (requireFields) {
200 MUSX_UNKNOWN_XML("xml element <" + fieldElement->getParent()->getTagName() + "> has child <" + fieldElement->getTagName() + "> which is not in the element list.");
201 }
202 }
203 }
204
205 static void populate(const std::shared_ptr<T>& instance, const XmlElementPtr& element)
206 {
207 if constexpr (std::is_base_of_v<TextsBase, T>) {
208 instance->text = element->getText();
209 } else {
210 for (auto child = element->getFirstChildElement(); child; child = child->getNextSibling()) {
211 populateField(instance, child);
212 }
213 }
214 }
215
216 template <typename... Args>
217 static std::shared_ptr<T> createAndPopulate(const XmlElementPtr& element, Args&&... args)
218 {
219 return FieldPopulator<T>::createAndPopulateImpl(element, std::forward<Args>(args)...);
220 }
221
223 template <typename... Args>
224 static std::shared_ptr<T> populateExistingOrCreate(
225 const XmlElementPtr& element, std::shared_ptr<T> instance, Args&&... args)
226 {
227 if (!instance) {
228 instance = std::make_shared<T>(std::forward<Args>(args)...);
229 }
230 FieldPopulator<T>::populate(instance, element);
231 return instance;
232 }
233
234private:
235 static const std::unordered_map<std::string_view, XmlElementPopulator<T>>& elementXref()
236 {
237 static const std::unordered_map<std::string_view, XmlElementPopulator<T>> xref = []()
238 {
239 std::unordered_map<std::string_view, XmlElementPopulator<T>> retval;
240 auto mappingArray = T::xmlMappingArray();
241 for (std::size_t i = 0; i < mappingArray.size(); i++) {
242 const XmlElementDescriptor<T> descriptor = mappingArray[i];
243 retval[std::get<0>(descriptor)] = std::get<1>(descriptor);
244 }
245 return retval;
246 }();
247 return xref;
248 }
249
250 template <typename... Args>
251 static std::shared_ptr<T> createAndPopulateImpl(const XmlElementPtr& element, Args&&... args)
252 {
253 auto instance = std::make_shared<T>(std::forward<Args>(args)...);
254 FieldPopulator<T>::populate(instance, element);
255 return instance;
256 }

265template <typename EnumClass, typename EmbeddedClass, typename... Args>
266inline void populateEmbeddedClass(const XmlElementPtr& e, std::unordered_map<EnumClass, std::shared_ptr<EmbeddedClass>>& listArray, Args&&... args)
267{
268 auto typeAttr = e->findAttribute("type");
269 if (!typeAttr) {
270 throw std::invalid_argument("<" + e->getTagName() + "> element has no type attribute");
271 }
272 listArray.emplace(toEnum<EnumClass>(typeAttr->getValueTrimmed()), FieldPopulator<EmbeddedClass>::createAndPopulate(e, std::forward<Args>(args)...));
273}

280template <typename T>
281inline std::vector<T> populateEmbeddedArray(const XmlElementPtr& e, const std::string_view& elementNodeName)
282{
283 std::vector<T> result;
284 for (auto child = e->getFirstChildElement(); child; child = child->getNextSibling()) {
285 if (child->getTagName() != elementNodeName) {
286 MUSX_UNKNOWN_XML("Unknown tag <" + child->getTagName() + "> while processing embedded xml array <" + e->getTagName() + ">");
287 continue;
288 }
289 if constexpr (std::is_fundamental_v<T>) {
290 result.push_back(child->getTextAs<T>());
291 } else if constexpr (std::is_same_v<T, std::string>) {
292 result.push_back(child->getText());
293 } else {
294 result.push_back(FieldPopulator<T>::createAndPopulate(child));
295 }
296 }
297 return result;
298}

302template <typename T>
303inline bool populateBoolean(const XmlElementPtr& element, const std::shared_ptr<T>& instance)
304{
305 MUSX_ASSERT_IF(!element) {
306 throw std::logic_error("Null element passed to populateBoolean function.");
307 }

316 } else {
317 return false; // I don't think we'll ever get an `offInPart` for the score, so assume it is for a part if we aren't a Base subclass
318 }

321inline std::vector<std::uint8_t> hexToBytes(std::string_view hex)
322{
323 std::vector<std::uint8_t> out;
324
325 if (hex.size() % 2 == 0) {
326 out.reserve(hex.size() / 2);
327 for (std::size_t i = 0; i < hex.size(); i += 2) {
328 unsigned value = 0;
329 const char* first = hex.data() + i;
330 const char* last = first + 2; // OK even when last == data()+size()
331 auto [ptr, ec] = std::from_chars(first, last, value, 16);
332 if (ec != std::errc()) {
333 i = hex.size(); // break out of loop if MUSX_INTEGRITY_ERROR does not throw
334 MUSX_INTEGRITY_ERROR("Invalid hex digit in hex string.");
335 }
336 out.push_back(static_cast<std::uint8_t>(value));
337 }
338 } else {
339 MUSX_INTEGRITY_ERROR("Encountered odd-length hex string.");
340 }
341
342 return out;
343}
