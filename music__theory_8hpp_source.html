137constexpr int calcDisplacement(const Pitch& pitch)
138{
139 int pitchClassVal = int(pitch.noteName) % STANDARD_DIATONIC_STEPS;
140 const int relativeOctave = pitch.octave - 4;
141
142 return pitchClassVal + (STANDARD_DIATONIC_STEPS * relativeOctave);
143}

148template <typename T>
149constexpr T sign(T n)
150{
151 static_assert(std::is_arithmetic_v<T>, "sign requires a numeric type");
152 return n < T(0) ? T(-1) : T(1);
153}
154
164template <typename T>
165constexpr T signedModulus(T n, T d)
166{
167 static_assert(std::is_integral_v<T>, "signedModulus requires an integer type");
168 return n % d;
169}
170
177template <typename T>
178constexpr T positiveModulus(T n, T d, T* q = nullptr)
179{
180 static_assert(std::is_integral_v<T>, "positiveModulus requires an integer type");
181 if (q) *q = n / d;
182 T result = signedModulus(n, d);
183 if (result < 0) {
184 result += d;
185 if (q) --(*q);
186 }
187 return result;
188}

205constexpr int calcPitchClass(NoteName noteName, int alteration = 0,
206 int numberOfEdoDivisions = STANDARD_12EDO_STEPS,
207 const std::array<int, STANDARD_DIATONIC_STEPS>& keyMap = MAJOR_KEYMAP)
208{
209 const int natural = keyMap[positiveModulus(int(noteName), STANDARD_DIATONIC_STEPS)];
210 return positiveModulus(natural + alteration, numberOfEdoDivisions);
211}

229constexpr int calc12EdoHalfstepsInInterval(int interval, int chromaticAlteration)
230{
231 int octaves{};
232 int diatonic = positiveModulus(interval, STANDARD_DIATONIC_STEPS, &octaves);
233 return MAJOR_KEYMAP[diatonic] + (octaves * STANDARD_12EDO_STEPS) + chromaticAlteration;
234}

240constexpr int calcAlterationFrom12EdoHalfsteps(int interval, int halfsteps)
241{
242 int octaves{};
243 int diatonic = positiveModulus(interval, STANDARD_DIATONIC_STEPS, &octaves);
244 int expectedHalfsteps = MAJOR_KEYMAP[diatonic] + (octaves * STANDARD_12EDO_STEPS);
245 return halfsteps - expectedHalfsteps;
246}

252constexpr int calcAlterationFromKeySigChange(int interval, int keySigChange)
253{
254 int diatonic = positiveModulus(interval, STANDARD_DIATONIC_STEPS);
255 int expectedKeyChange = DIATONIC_INTERVAL_ADJUSTMENTS[diatonic][0];
256 if (interval < 0) {
257 if (std::abs(expectedKeyChange) > 1) { // imperfect intervals
258 expectedKeyChange -= STANDARD_DIATONIC_STEPS;
259 }
260 }
261 int alteration = (keySigChange - expectedKeyChange) / STANDARD_DIATONIC_STEPS;
262 return alteration;
263}

271constexpr int calcKeySigChangeFromInterval(int interval, int chromaticAlteration)
272{
273 const int diatonic = positiveModulus(interval, STANDARD_DIATONIC_STEPS);
274 int expectedKeyChange = DIATONIC_INTERVAL_ADJUSTMENTS[diatonic][0];
275 if (interval < 0) {
276 if (std::abs(expectedKeyChange) > 1) { // imperfect intervals
277 expectedKeyChange -= STANDARD_DIATONIC_STEPS;
278 }
279 }
280 return expectedKeyChange + (chromaticAlteration * STANDARD_DIATONIC_STEPS);
281}

298class Transposer
299{
300private:
301 int m_displacement;
302 int m_alteration; // alteration from key signature
303 int m_numberOfEdoDivisions; // number of divisions in the EDO (default 12)
304 std::vector<int> m_keyMap; // step map for the EDO
305
306public:
309 explicit Transposer(const Pitch& pitch)
310 : Transposer(calcDisplacement(pitch), pitch.alteration)
311 {
312 }
313
322 Transposer(int displacement, int alteration,
323 bool isMinor = false, int numberOfEdoDivisions = STANDARD_12EDO_STEPS,
324 const std::optional <std::vector<int>>& keyMap = std::nullopt)
325 : m_displacement(displacement), m_alteration(alteration), m_numberOfEdoDivisions(numberOfEdoDivisions)
326 {
327 if (keyMap) {
328 if (keyMap.value().size() != STANDARD_DIATONIC_STEPS) {
329 throw std::invalid_argument("The Transposer class only supports key map arrays of " + std::to_string(STANDARD_DIATONIC_STEPS) + " elements");
330 }
331 m_keyMap = keyMap.value();
332 } else if (isMinor) {
333 m_keyMap.assign(MINOR_KEYMAP.begin(), MINOR_KEYMAP.end());
334 } else {
335 m_keyMap.assign(MAJOR_KEYMAP.begin(), MAJOR_KEYMAP.end());
336 }
337 }

340 int displacement() const { return m_displacement; }
341
343 int alteration() const { return m_alteration; }

347 void diatonicTranspose(int interval)
348 {
349 m_displacement += interval;
350 }

354 void enharmonicTranspose(int diatonicSteps)
355 {
356 const int stepSign = sign(diatonicSteps);
357 for (int i = 0; i < std::abs(diatonicSteps); ++i) {
358 const int keyStepEnharmonic = calcStepsBetweenScaleDegrees(m_displacement, m_displacement + stepSign);
359 diatonicTranspose(stepSign);
360 m_alteration -= stepSign * keyStepEnharmonic;
361 }
362 }

385 void chromaticTranspose(int interval, int chromaticAlteration)
386 {
387 const int intervalNormalized = signedModulus(interval, STANDARD_DIATONIC_STEPS);
388 const int stepsInAlteration = calcStepsInAlteration(interval, chromaticAlteration);
389 const int stepsInInterval = calcStepsInNormalizedInterval(intervalNormalized);
390 const int stepsInDiatonicInterval = calcStepsBetweenScaleDegrees(m_displacement, m_displacement + intervalNormalized);
391
392 const int effectiveAlteration = stepsInAlteration + stepsInInterval - sign(interval) * stepsInDiatonicInterval;
393
394 diatonicTranspose(interval);
395 m_alteration += effectiveAlteration;
396 }

404 void simplifySpelling()
405 {
406 while (std::abs(m_alteration) > 0) {
407 const int currSign = sign(m_alteration);
408 const int currAbsDisp = std::abs(m_alteration);
409 enharmonicTranspose(currSign);
410 if (std::abs(m_alteration) >= currAbsDisp) {
411 enharmonicTranspose(-currSign);
412 return;
413 }
414 if (currSign != sign(m_alteration)) {
415 break;
416 }
417 }
418 }

429 void stepwiseTranspose(int numberOfEdoDivisions)
430 {
431 m_alteration += numberOfEdoDivisions;
432 simplifySpelling();
433 }

445 bool isEnharmonicEquivalent(int displacement, int alteration) const {
446 return calcAbsoluteDivision(displacement, alteration) == calcAbsoluteDivision(m_displacement, m_alteration);
447 }
448
449private:
450 int calcFifthSteps() const
451 {
452 // std::log(3.0 / 2.0) / std::log(2.0) is 0.5849625007211562.
453 static constexpr double kFifthsMultiplier = 0.5849625007211562;
454 return static_cast<int>(std::floor(m_numberOfEdoDivisions * kFifthsMultiplier) + 0.5);
455 }
456
457 int calcScaleDegree(int interval) const
458 { return positiveModulus(interval, int(m_keyMap.size())); }
459
460 int calcStepsBetweenScaleDegrees(int firstDisplacement, int secondDisplacement) const
461 {
462 const int firstScaleDegree = calcScaleDegree(firstDisplacement);
463 const int secondScaleDegree = calcScaleDegree(secondDisplacement);
464 int result = sign(secondDisplacement - firstDisplacement) * (m_keyMap[secondScaleDegree] - m_keyMap[firstScaleDegree]);
465 if (result < 0) {
466 result += m_numberOfEdoDivisions;
467 }
468 return result;
469 }
470
471 int calcStepsInAlteration(int interval, int alteration) const
472 {
473 const int fifthSteps = calcFifthSteps();
474 const int plusFifths = sign(interval) * alteration * 7; // number of fifths to add for a chromatic halfstep alteration (in any EDO)
475 const int minusOctaves = sign(interval) * alteration * -4; // number of octaves to subtract for a chromatic halfstep alteration (in any EDO)
476 const int result = sign(interval) * ((plusFifths * fifthSteps) + (minusOctaves * m_numberOfEdoDivisions));
477 return result;
478 }
479
480 int calcStepsInNormalizedInterval(int intervalNormalized) const
481 {
482 const int fifthSteps = calcFifthSteps();
483 const int index = std::abs(intervalNormalized);
484 const int plusFifths = DIATONIC_INTERVAL_ADJUSTMENTS[index][0]; // number of fifths
485 const int minusOctaves = DIATONIC_INTERVAL_ADJUSTMENTS[index][1]; // number of octaves
486
487 return sign(intervalNormalized) * ((plusFifths * fifthSteps) + (minusOctaves * m_numberOfEdoDivisions));
488 }
489
490 int calcAbsoluteDivision(int displacement, int alteration) const {
491 const int scaleDegree = calcScaleDegree(displacement); // 0..6
492 const int baseStep = m_keyMap[scaleDegree];
493
494 const int octaveCount = (displacement < 0 && displacement % STANDARD_DIATONIC_STEPS != 0)
495 ? (displacement / STANDARD_DIATONIC_STEPS) - 1
496 : (displacement / STANDARD_DIATONIC_STEPS);
497 const int octaveSteps = octaveCount * m_numberOfEdoDivisions;
498 const int chromaticSteps = calcStepsInAlteration(/*interval=*/+1, alteration);
499
500 return baseStep + chromaticSteps + octaveSteps;
501 }
