197class ContainedClassBase : public EnigmaBase
198{
199 Cmper getSourcePartId() const = delete;
200
201protected:
202 // A shallow copy of an owning OthersBase or DetailsBase leaves contained objects parented to
203 // the source instance. If that distinction matters, the owner must use PartContextRebinder.
204 // PartContextRebinder<details::CenterShape> is a simple example.
205
207 ContainedClassBase& operator=(const ContainedClassBase& other)
208 {
209 if (this != &other) {
210 this->EnigmaBase::operator=(other);
211 }
212 return *this;
213 }
214
216 ContainedClassBase& operator=(ContainedClassBase&& other) noexcept
217 {
218 if (this != &other) {
219 this->EnigmaBase::operator=(std::move(other));
220 }
221 return *this;
222 }
223
224public:
230 ContainedClassBase(const MusxInstance<EnigmaBase>& parent)
231 : EnigmaBase(parent->getDocument(), SCORE_PARTID, ShareMode::All), m_parent(parent)
232 {}
233
235 template <typename ParentClass = EnigmaBase>
236 MusxInstance<ParentClass> getParent() const
237 {
238 auto result = m_parent.lock();
239 MUSX_ASSERT_IF (!result) {
240 throw std::logic_error("Attempt to get parent of contained class, but the parent is no longer allocated.");
241 }
242 if constexpr (std::is_same_v<EnigmaBase, ParentClass>) {
243 return result;
244 } else {
245 return std::dynamic_pointer_cast<const ParentClass>(result);
246 }
247 }
248
249private:
250 const MusxInstanceWeak<EnigmaBase> m_parent;
251};
